## 验收证据总表（一次跑完）

```bash
cd <你的工作目录>/cc-haha-android

echo "=== P0-2 .gitignore 生效验证 ==="
git status --short | grep -E "\.pkg$|\.apk$|node_modules|/build/" || echo "✅ 无大文件入库"

echo "=== P0-3 仓库体积 ==="
git count-objects -vH

echo "=== P0-1 密钥扫描 ==="
node tools/check-secrets.mjs && echo "✅ 无密钥"

echo "=== P0-5 上游未入库 ==="
git ls-files | grep -c "runtime/upstream/" | grep -q "^0$" && echo "✅ upstream 未入库"

echo "=== P0-6 README 声明 ==="
head -30 README.md | grep -E "非官方|无任何关联|风险披露" && echo "✅ 声明就位"

echo "=== P1-1 签名 ==="
"$ANDROID_HOME/build-tools/34.0.0/apksigner" verify --print-certs \
  android/app/build/outputs/apk/release/app-release.apk | head -5 |

echo "=== P1-14 校验和 ==="
cd dist && sha256sum -c SHA256SUMS
```
